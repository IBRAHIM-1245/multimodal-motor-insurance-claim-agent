# Database
psycopg2-binary   # connect Python to Postgres (Neon)
python-dotenv     # read keys from the .env file

# Claim classifier (install torch first with the CPU command in the README)
torch             # runs the model
transformers      # MuRIL model and tokenizer
peft              # loads our LoRA adapter

# Agent
langgraph         # builds the step-by-step claim agent (graph of nodes)
langchain-openai  # LangChain wrapper for OpenAI chat models and embeddings
pinecone          # vector database client (policy rule search)

# API
fastapi           # web API framework for the agent service
uvicorn           # server that runs the FastAPI app

# Monitoring
langfuse          # traces every agent step, LLM call, tokens and cost
